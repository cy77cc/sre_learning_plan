#### 2.1 Python 应用指标采集

```python
# app_metrics.py - Flask 应用集成 Prometheus 指标
from flask import Flask, request, Response
from prometheus_client import (
    Counter, Histogram, Gauge, Info,
    generate_latest, CONTENT_TYPE_LATEST,
    CollectorRegistry, multiprocess
)
import time
import random
import os

app = Flask(__name__)

# ===== 定义指标 =====

# 请求计数器 (RED - Rate)
REQUEST_COUNT = Counter(
    'http_requests_total',
    'Total HTTP requests',
    ['method', 'endpoint', 'status_code']
)

# 请求延迟直方图 (RED - Duration)
REQUEST_LATENCY = Histogram(
    'http_request_duration_seconds',
    'HTTP request latency in seconds',
    ['method', 'endpoint'],
    buckets=[0.01, 0.025, 0.05, 0.1, 0.25, 0.5, 1.0, 2.5, 5.0, 10.0]
)

# 错误计数器 (RED - Errors)
ERROR_COUNT = Counter(
    'http_errors_total',
    'Total HTTP errors',
    ['method', 'endpoint', 'status_code', 'error_type']
)

# 活跃请求数
ACTIVE_REQUESTS = Gauge(
    'http_active_requests',
    'Number of active HTTP requests',
    ['method']
)

# 业务指标 - 订单相关
ORDER_COUNT = Counter(
    'orders_total',
    'Total orders created',
    ['status', 'payment_method']
)

ORDER_AMOUNT = Histogram(
    'order_amount_dollars',
    'Order amount in dollars',
    buckets=[10, 25, 50, 100, 250, 500, 1000, 2500, 5000]
)

# 业务指标 - 用户相关
USER_LOGIN_COUNT = Counter(
    'user_logins_total',
    'Total user logins',
    ['method', 'status']
)

ACTIVE_USERS = Gauge(
    'active_users',
    'Number of currently active users'
)

# 应用信息
APP_INFO = Info(
    'app',
    'Application information'
)
APP_INFO.info({
    'version': os.getenv('APP_VERSION', '1.0.0'),
    'environment': os.getenv('ENVIRONMENT', 'development'),
    'team': 'backend'
})

# ===== 中间件 =====

@app.before_request
def before_request():
    """请求前: 记录开始时间，增加活跃请求"""
    request._start_time = time.time()
    ACTIVE_REQUESTS.labels(method=request.method).inc()

@app.after_request
def after_request(response):
    """请求后: 记录指标"""
    # 计算延迟
    latency = time.time() - getattr(request, '_start_time', time.time())

    # 记录请求计数
    REQUEST_COUNT.labels(
        method=request.method,
        endpoint=request.path,
        status_code=response.status_code
    ).inc()

    # 记录延迟
    REQUEST_LATENCY.labels(
        method=request.method,
        endpoint=request.path
    ).observe(latency)

    # 记录错误
    if response.status_code >= 400:
        ERROR_COUNT.labels(
            method=request.method,
            endpoint=request.path,
            status_code=response.status_code,
            error_type='client' if response.status_code < 500 else 'server'
        ).inc()

    # 减少活跃请求
    ACTIVE_REQUESTS.labels(method=request.method).dec()

    return response

# ===== 路由 =====

@app.route('/metrics')
def metrics():
    """Prometheus 指标端点"""
    return Response(
        generate_latest(),
        mimetype=CONTENT_TYPE_LATEST
    )

@app.route('/api/orders', methods=['POST'])
def create_order():
    """创建订单 - 业务指标示例"""
    # 模拟订单处理
    time.sleep(random.uniform(0.05, 0.3))

    # 模拟订单结果
    status = random.choices(
        ['success', 'failed', 'cancelled'],
        weights=[85, 10, 5]
    )[0]

    payment_method = random.choice(['credit_card', 'alipay', 'wechat'])

    # 记录业务指标
    ORDER_COUNT.labels(
        status=status,
        payment_method=payment_method
    ).inc()

    amount = random.uniform(10, 500)
    ORDER_AMOUNT.observe(amount)

    if status == 'success':
        return {'status': 'success', 'order_id': 'ORD-12345', 'amount': amount}
    elif status == 'failed':
        return {'status': 'failed', 'error': 'payment_failed'}, 400
    else:
        return {'status': 'cancelled'}, 499

@app.route('/api/login', methods=['POST'])
def user_login():
    """用户登录 - 业务指标示例"""
    time.sleep(random.uniform(0.1, 0.5))

    success = random.random() < 0.9
    USER_LOGIN_COUNT.labels(
        method='password',
        status='success' if success else 'failed'
    ).inc()

    if success:
        ACTIVE_USERS.inc()
        return {'status': 'success'}
    return {'status': 'failed'}, 401

@app.route('/api/users')
def get_users():
    """获取用户列表 - 模拟业务接口"""
    time.sleep(random.uniform(0.01, 0.1))
    return {'users': ['user1', 'user2', 'user3']}

@app.route('/healthz')
def health():
    """健康检查"""
    return {'status': 'healthy'}

@app.route('/readyz')
def ready():
    """就绪检查"""
    return {'status': 'ready'}

if __name__ == '__main__':
    ACTIVE_USERS.set(random.randint(100, 500))
    app.run(host='0.0.0.0', port=8080)
```

#### 2.2 Go 应用指标采集

```go
// main.go - Go HTTP 应用集成 Prometheus 指标
package main

import (
    "fmt"
    "math/rand"
    "net/http"
    "os"
    "time"

    "github.com/prometheus/client_golang/prometheus"
    "github.com/prometheus/client_golang/prometheus/promauto"
    "github.com/prometheus/client_golang/prometheus/promhttp"
)

// ===== 定义指标 =====

var (
    // RED 方法指标
    httpRequestsTotal = promauto.NewCounterVec(
        prometheus.CounterOpts{
            Name: "http_requests_total",
            Help: "Total number of HTTP requests",
        },
        []string{"method", "path", "status"},
    )

    httpRequestDuration = promauto.NewHistogramVec(
        prometheus.HistogramOpts{
            Name:    "http_request_duration_seconds",
            Help:    "HTTP request duration in seconds",
            Buckets: []float64{0.01, 0.025, 0.05, 0.1, 0.25, 0.5, 1.0, 2.5, 5.0},
        },
        []string{"method", "path"},
    )

    httpErrorsTotal = promauto.NewCounterVec(
        prometheus.CounterOpts{
            Name: "http_errors_total",
            Help: "Total number of HTTP errors",
        },
        []string{"method", "path", "status", "type"},
    )

    // 活跃请求
    activeRequests = promauto.NewGauge(
        prometheus.GaugeOpts{
            Name: "http_active_requests",
            Help: "Number of active HTTP requests",
        },
    )

    // 业务指标
    ordersTotal = promauto.NewCounterVec(
        prometheus.CounterOpts{
            Name: "orders_total",
            Help: "Total number of orders",
        },
        []string{"status", "payment_method"},
    )

    orderAmount = promauto.NewHistogram(
        prometheus.HistogramOpts{
            Name:    "order_amount_dollars",
            Help:    "Order amount in dollars",
            Buckets: []float64{10, 25, 50, 100, 250, 500, 1000, 2500, 5000},
        },
    )

    // 数据库查询延迟
    dbQueryDuration = promauto.NewHistogramVec(
        prometheus.HistogramOpts{
            Name:    "db_query_duration_seconds",
            Help:    "Database query duration in seconds",
            Buckets: []float64{0.001, 0.005, 0.01, 0.025, 0.05, 0.1, 0.25, 0.5},
        },
        []string{"operation", "table"},
    )

    // 应用版本信息
    appInfo = promauto.NewGaugeVec(
        prometheus.GaugeOpts{
            Name: "app_info",
            Help: "Application information",
        },
        []string{"version", "environment", "team"},
    )
)

// ===== 中间件 =====

func metricsMiddleware(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        start := time.Now()
        activeRequests.Inc()
        defer activeRequests.Dec()

        // 包装 ResponseWriter 以捕获状态码
        wrapped := &responseWriter{ResponseWriter: w, statusCode: http.StatusOK}
        next.ServeHTTP(wrapped, r)

        duration := time.Since(start).Seconds()

        // 记录指标
        status := fmt.Sprintf("%d", wrapped.statusCode)
        httpRequestsTotal.WithLabelValues(r.Method, r.URL.Path, status).Inc()
        httpRequestDuration.WithLabelValues(r.Method, r.URL.Path).Observe(duration)

        if wrapped.statusCode >= 400 {
            errorType := "client"
            if wrapped.statusCode >= 500 {
                errorType = "server"
            }
            httpErrorsTotal.WithLabelValues(r.Method, r.URL.Path, status, errorType).Inc()
        }
    })
}

type responseWriter struct {
    http.ResponseWriter
    statusCode int
}

func (rw *responseWriter) WriteHeader(code int) {
    rw.statusCode = code
    rw.ResponseWriter.WriteHeader(code)
}

// ===== 处理器 =====

func createOrderHandler(w http.ResponseWriter, r *http.Request) {
    time.Sleep(time.Duration(rand.Intn(300)) * time.Millisecond)

    statuses := []string{"success", "failed", "cancelled"}
    weights := []int{85, 10, 5}
    status := weightedChoice(statuses, weights)

    paymentMethods := []string{"credit_card", "alipay", "wechat"}
    payment := paymentMethods[rand.Intn(len(paymentMethods))]

    ordersTotal.WithLabelValues(status, payment).Inc()
    amount := 10 + rand.Float64()*490
    orderAmount.Observe(amount)

    w.Header().Set("Content-Type", "application/json")
    fmt.Fprintf(w, `{"status":"%s","amount":%.2f}`, status, amount)
}

func healthHandler(w http.ResponseWriter, r *http.Request) {
    w.Header().Set("Content-Type", "application/json")
    w.Write([]byte(`{"status":"healthy"}`))
}

func weightedChoice(items []string, weights []int) string {
    total := 0
    for _, w := range weights {
        total += w
    }
    r := rand.Intn(total)
    for i, w := range weights {
        r -= w
        if r < 0 {
            return items[i]
        }
    }
    return items[0]
}

func main() {
    // 设置应用信息
    version := os.Getenv("APP_VERSION")
    if version == "" {
        version = "1.0.0"
    }
    environment := os.Getenv("ENVIRONMENT")
    if environment == "" {
        environment = "development"
    }
    appInfo.WithLabelValues(version, environment, "backend").Set(1)

    mux := http.NewServeMux()

    // 业务路由
    mux.HandleFunc("/api/orders", createOrderHandler)
    mux.HandleFunc("/healthz", healthHandler)

    // Prometheus 指标端点
    mux.Handle("/metrics", promhttp.Handler())

    // 应用中间件
    handler := metricsMiddleware(mux)

    fmt.Println("Server starting on :8080")
    if err := http.ListenAndServe(":8080", handler); err != nil {
        fmt.Printf("Server failed: %v\n", err)
        os.Exit(1)
    }
}
```

#### 3.2 PrometheusRule 告警规则

```yaml
# k8s/prometheus-rules.yaml
apiVersion: monitoring.coreos.com/v1
kind: PrometheusRule
metadata:
  name: demo-app-alerts
  namespace: monitoring
  labels:
    release: kube-prometheus-stack
spec:
  groups:
    - name: demo-app.rules
      rules:
        # 记录规则 - 预计算复杂查询
        - record: demo_app:http_requests:rate5m
          expr: |
            sum(rate(http_requests_total{job="demo-app"}[5m])) by (method, path)

        - record: demo_app:http_errors:rate5m
          expr: |
            sum(rate(http_errors_total{job="demo-app"}[5m])) by (method, path)

        - record: demo_app:http_request_duration:p99
          expr: |
            histogram_quantile(0.99, sum(rate(http_request_duration_seconds_bucket{job="demo-app"}[5m])) by (le, path))

    - name: demo-app.alerts
      rules:
        # 高错误率告警
        - alert: DemoAppHighErrorRate
          expr: |
            sum(rate(http_errors_total{job="demo-app", type="server"}[5m]))
            / sum(rate(http_requests_total{job="demo-app"}[5m])) > 0.05
          for: 5m
          labels:
            severity: critical
            team: backend
            service: demo-app
          annotations:
            summary: "Demo App 错误率过高"
            description: |
              当前错误率: {{ $value | humanizePercentage }}
              超过 5% 阈值，请立即排查。

        # 高延迟告警
        - alert: DemoAppHighLatency
          expr: |
            histogram_quantile(0.99, sum(rate(http_request_duration_seconds_bucket{job="demo-app"}[5m])) by (le)) > 1
          for: 5m
          labels:
            severity: warning
            team: backend
            service: demo-app
          annotations:
            summary: "Demo App P99 延迟过高"
            description: "P99 延迟: {{ $value }}s，超过 1s 阈值。"

        # 请求量下降告警
        - alert: DemoAppLowTraffic
          expr: |
            sum(rate(http_requests_total{job="demo-app"}[5m])) < 0.1
          for: 15m
          labels:
            severity: warning
            team: backend
            service: demo-app
          annotations:
            summary: "Demo App 请求量异常下降"
            description: "当前 QPS: {{ $value }}，可能服务不可用或流量下降。"
```

## 面试题精选

### 1. 如何设计一个完整的监控体系？需要包含哪些层次？

**参考答案：**

完整监控体系应包含四个层次：

1. **采集层**：node-exporter（主机）、kube-state-metrics（K8s状态）、cAdvisor（容器）、应用自定义指标（client library）
2. **存储层**：Prometheus（指标）、Loki（日志）、Jaeger/Tempo（链路追踪）
3. **展示层**：Grafana，按基础设施/应用/业务三层设计仪表盘
4. **告警层**：Alertmanager，支持多渠道路由、分组、抑制、静默

设计原则：分层设计、告警分级、关联追踪、SLO 驱动。

### 2. 如何在应用中暴露自定义 Prometheus 指标？

**参考答案：**

步骤：

1. 选择客户端库：Python 用 `prometheus_client`，Go 用 `prometheus/client_golang`
2. 定义指标类型：Counter（递增）、Gauge（可增可减）、Histogram（分布）、Summary（分位数）
3. 在代码中记录指标：HTTP 中间件记录请求指标，业务逻辑记录业务指标
4. 暴露 `/metrics` 端点
5. 在 K8s 中配置 ServiceMonitor 或 PodMonitor
6. 确保 Prometheus 能通过 Service 发现 Pod

### 3. Alertmanager 的路由机制是怎样的？

**参考答案：**

Alertmanager 路由是树形结构：

1. 告警从根路由开始匹配
2. 按 `matchers`（标签匹配）选择路由
3. 匹配到的第一个路由处理告警（除非 `continue: true`）
4. 每个路由可配置：receiver（接收者）、group_by（分组标签）、group_wait（等待时间）、repeat_interval（重复间隔）
5. 支持抑制规则：高优先级告警可抑制低优先级
6. 支持静默规则：临时屏蔽特定告警

### 4. RED 方法和 USE 方法分别适用于什么场景？

**参考答案：**

- **RED 方法**（Rate, Errors, Duration）：面向请求的服务。Rate = 请求速率，Errors = 错误率，Duration = 请求延迟。适用于 API 服务、Web 应用等。

- **USE 方法**（Utilization, Saturation, Errors）：基础设施资源。Utilization = 资源使用率，Saturation = 饱和度（排队），Errors = 资源错误。适用于 CPU、内存、磁盘、网络等。

选择依据：面向用户的服务用 RED，面向资源的监控用 USE。完整的监控体系两者都要覆盖。

### 5. 如何确保告警的有效性，避免告警疲劳？

**参考答案：**

避免告警疲劳的策略：

1. **SLO 驱动告警**：基于用户可感知的 SLI 设置告警，而非技术指标
2. **多窗口燃烧速率**：使用长窗口低倍数 + 短窗口高倍数，减少噪音
3. **告警分级**：Critical（立即响应）、Warning（工作时间处理）、Info（记录）
4. **告警抑制**：Critical 告警触发时抑制同一服务的 Warning
5. **告警聚合**：通过 group_by 聚合相关告警
6. **定期评审**：每月评审告警规则，删除无用告警
7. **告警静默**：维护窗口使用静默规则

### 6. 如何在 Grafana 中实现三层仪表盘的下钻？

**参考答案：**

下钻实现方式：

1. **变量传递**：使用 Grafana 变量（如 `$service`、`$instance`）在不同层级间传递上下文
2. **面板链接**：在面板中配置 Data Link，点击后跳转到下一层仪表盘并传递变量
3. **仪表盘链接**：在仪表盘设置中配置 Dashboard Links，建立层级关系
4. **行折叠**：使用 Row 组织面板，通过展开/折叠控制详细程度

示例：Overview 点击服务名 → Service Dashboard（自动填充 $service 变量）→ 点击实例 → Instance Dashboard。

### 7. Prometheus 的 Recording Rules 有什么作用？何时使用？

**参考答案：**

Recording Rules 预计算 PromQL 表达式结果并保存为新的时间序列。

使用场景：
1. **复杂查询优化**：将多次使用的复杂查询预计算，减少 Dashboard 加载时间
2. **告警规则优化**：告警规则中引用预计算结果，提高评估效率
3. **跨时间范围聚合**：如计算小时级、天级聚合指标

```yaml
groups:
  - name: recording_rules
    rules:
      - record: instance:http_requests:rate5m
        expr: sum(rate(http_requests_total[5m])) by (instance)
```

### 8. 如何处理 Prometheus 高可用和长期存储？

**参考答案：**

高可用方案：
1. **Prometheus 双副本**：运行两个相同配置的 Prometheus，Alertmanager 自动去重
2. **Thanos Sidecar**：每个 Prometheus 附带 Thanos Sidecar，上传数据到对象存储
3. **Thanos Receive**：通过 Remote Write 接收多个 Prometheus 的数据
4. **Mimir/Cortex**：多租户、长期存储的 Prometheus 兼容方案

长期存储：
1. **Thanos Store**：从对象存储查询历史数据
2. **Remote Write**：将数据写入 InfluxDB、Thanos Receive 等
3. **VictoriaMetrics**：高压缩比的长期存储方案

### 9. 如何设计告警通知模板，使告警信息更有价值？

**参考答案：**

好的告警通知应包含：

1. **标题**：简洁描述问题（如 "[Critical] API Server Error Rate > 5%"）
2. **摘要**：一句话说明问题
3. **描述**：详细说明影响和当前值
4. **标签**：severity、namespace、service、instance
5. **链接**：Grafana Dashboard 链接、Runbook 链接
6. **恢复信息**：告警恢复时发送 resolved 通知

```
*Alert:* Demo App 错误率过高
*Status:* FIRING
*Severity:* critical
*Namespace:* production
*Service:* demo-app
*Current Value:* 8.5% (阈值: 5%)
*Dashboard:* http://grafana/d/xxx
*Runbook:* https://wiki/runbook/xxx
```

### 10. 如何监控微服务架构中的服务依赖关系？

**参考答案：**

监控微服务依赖关系的方法：

1. **链路追踪**：使用 Jaeger/Zipkin/Tempo 追踪请求流经的服务链路
2. **服务拓扑图**：使用 Grafana Node Graph 面板展示服务依赖
3. **依赖健康指标**：为每个下游服务单独监控延迟和错误率
4. **服务地图**：基于 RED 方法构建服务间调用关系图
5. **关键路径监控**：识别关键业务路径，设置端到端 SLI

```promql
# 服务间调用延迟
sum(rate(http_client_request_duration_seconds_bucket{target_service="payment"}[5m])) by (le)
```
