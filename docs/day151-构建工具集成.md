### 2. Make

#### 2.1 Makefile 基础

```makefile
# Makefile - Go 项目示例

# ── 变量定义 ──────────────────────────────────────────────
APP_NAME    := myapp
VERSION     := $(shell git describe --tags --always --dirty)
BUILD_TIME  := $(shell date -u '+%Y-%m-%dT%H:%M:%SZ')
GO_VERSION  := 1.22
LDFLAGS     := -X main.version=$(VERSION) -X main.buildTime=$(BUILD_TIME)

# Go 源文件
GO_SOURCES  := $(shell find . -name '*.go' -not -path './vendor/*')
GO_PACKAGES := $(shell go list ./...)

# Docker
REGISTRY    := registry.example.com
IMAGE       := $(REGISTRY)/$(APP_NAME):$(VERSION)

# ── 默认目标 ─────────────────────────────────────────────
.PHONY: all
all: lint test build

# ── 构建目标 ─────────────────────────────────────────────
.PHONY: build
build: ## 构建应用
	@echo "Building $(APP_NAME) $(VERSION)..."
	CGO_ENABLED=0 go build \
		-ldflags "$(LDFLAGS)" \
		-o bin/$(APP_NAME) \
		./cmd/server

.PHONY: build-all
build-all: ## 交叉编译所有平台
	@echo "Cross-compiling..."
	GOOS=linux GOARCH=amd64 go build -ldflags "$(LDFLAGS)" -o bin/$(APP_NAME)-linux-amd64 ./cmd/server
	GOOS=linux GOARCH=arm64 go build -ldflags "$(LDFLAGS)" -o bin/$(APP_NAME)-linux-arm64 ./cmd/server
	GOOS=darwin GOARCH=amd64 go build -ldflags "$(LDFLAGS)" -o bin/$(APP_NAME)-darwin-amd64 ./cmd/server
	GOOS=darwin GOARCH=arm64 go build -ldflags "$(LDFLAGS)" -o bin/$(APP_NAME)-darwin-arm64 ./cmd/server

# ── 测试目标 ─────────────────────────────────────────────
.PHONY: test
test: ## 运行单元测试
	@echo "Running tests..."
	go test -race -coverprofile=coverage.out ./...

.PHONY: test-integration
test-integration: ## 运行集成测试
	@echo "Running integration tests..."
	go test -tags=integration -race ./...

.PHONY: coverage
coverage: test ## 生成覆盖率报告
	@echo "Coverage: $(shell go tool cover -func=coverage.out | tail -1 | awk '{print $$3}')"
	go tool cover -html=coverage.out -o coverage.html

# ── 代码质量 ─────────────────────────────────────────────
.PHONY: lint
lint: ## 代码检查
	@echo "Running linters..."
	golangci-lint run --timeout 5m

.PHONY: fmt
fmt: ## 格式化代码
	@echo "Formatting..."
	gofmt -s -w .
	goimports -w .

.PHONY: vet
vet: ## Go vet 检查
	go vet ./...

# ── Docker 目标 ──────────────────────────────────────────
.PHONY: docker-build
docker-build: ## 构建 Docker 镜像
	@echo "Building Docker image $(IMAGE)..."
	docker build -t $(IMAGE) .
	docker tag $(IMAGE) $(REGISTRY)/$(APP_NAME):latest

.PHONY: docker-push
docker-push: docker-build ## 推送 Docker 镜像
	@echo "Pushing $(IMAGE)..."
	docker push $(IMAGE)
	docker push $(REGISTRY)/$(APP_NAME):latest

.PHONY: docker-run
docker-run: ## 运行 Docker 容器
	docker run --rm -p 8080:8080 $(IMAGE)

# ── 部署目标 ─────────────────────────────────────────────
.PHONY: deploy
deploy: ## 部署到 Kubernetes
	@echo "Deploying $(IMAGE) to $(DEPLOY_ENV)..."
	kubectl set image deployment/$(APP_NAME) $(APP_NAME)=$(IMAGE)
	kubectl rollout status deployment/$(APP_NAME) --timeout=300s

.PHONY: deploy-staging
deploy-staging: ## 部署到 staging
	$(MAKE) deploy DEPLOY_ENV=staging

.PHONY: deploy-production
deploy-production: ## 部署到 production
	$(MAKE) deploy DEPLOY_ENV=production

# ── 清理目标 ─────────────────────────────────────────────
.PHONY: clean
clean: ## 清理构建产物
	@echo "Cleaning..."
	rm -rf bin/
	rm -rf dist/
	rm -f coverage.out coverage.html
	go clean -cache

# ── 依赖管理 ─────────────────────────────────────────────
.PHONY: deps
deps: ## 下载依赖
	go mod download

.PHONY: deps-update
deps-update: ## 更新依赖
	go get -u ./...
	go mod tidy

.PHONY: deps-vendor
deps-vendor: ## 更新 vendor 目录
	go mod vendor

# ── 开发工具 ─────────────────────────────────────────────
.PHONY: dev
dev: ## 启动开发服务器（热重载）
	air -c .air.toml

.PHONY: generate
generate: ## 运行代码生成
	go generate ./...

# ── 帮助 ─────────────────────────────────────────────────
.PHONY: help
help: ## 显示帮助
	@grep -E '^[a-zA-Z_-]+:.*?## .*$$' $(MAKEFILE_LIST) | sort | \
		awk 'BEGIN {FS = ":.*?## "}; {printf "\033[36m%-20s\033[0m %s\n", $$1, $$2}'
```

#### 2.2 Make 高级特性

```makefile
# 条件判断
ifeq ($(ENV),production)
    LDFLAGS += -s -w              # 去除调试信息
    GOFLAGS += -trimpath
else
    GOFLAGS += -race              # 开发环境启用竞态检测
endif

# 函数
SOURCES = $(wildcard src/*.go)
OBJECTS = $(SOURCES:src/%.go=bin/%)

# 模式规则
bin/%: src/%.go
	go build -o $@ $<

# 自动变量
# $@ - 目标文件名
# $< - 第一个依赖文件名
# $^ - 所有依赖文件名
# $? - 比目标新的依赖文件

# 多目标依赖
build test lint: deps

# 并行执行
# make -j4 test    # 4 个并行任务

# 包含其他 Makefile
-include .env
include mk/*.mk

# 多行命令 + 错误处理
.PHONY: deploy
deploy:
	@set -e; \
	echo "Deploying..."; \
	kubectl apply -f k8s/; \
	kubectl rollout status deployment/$(APP_NAME)
```

### 3. Bazel

#### 3.1 Bazel 概述

```
Bazel 架构：

┌──────────────────────────────────────────────────────────────┐
│                      Bazel 架构                               │
├──────────────────────────────────────────────────────────────┤
│                                                              │
│  ┌──────────────────────────────────────────────────────┐   │
│  │                    Bazel Server                        │   │
│  │                                                      │   │
│  │  ┌──────────┐  ┌──────────┐  ┌──────────┐           │   │
│  │  │ Loading  │  │ Analysis │  │Execution │           │   │
│  │  │ Phase    │  │ Phase    │  │Phase     │           │   │
│  │  └──────────┘  └──────────┘  └──────────┘           │   │
│  │                                                      │   │
│  │  Loading:  解析 BUILD 文件，加载规则                   │   │
│  │  Analysis: 分析依赖图，确定构建计划                    │   │
│  │  Execution: 执行构建动作                              │   │
│  └──────────────────────────────────────────────────────┘   │
│                                                              │
│  核心特性：                                                  │
│  • 增量构建（只重建变更部分）                                 │
│  • 远程缓存（团队共享构建缓存）                               │
│  • 远程执行（分布式构建）                                    │
│  • 可重复性（沙箱隔离）                                      │
│  • 多语言支持（Java, Go, Python, C++, Rust 等）              │
│  • 单体仓库友好                                              │
└──────────────────────────────────────────────────────────────┘
```

#### 6.2 Go 构建优化

```makefile
# Makefile - Go 构建优化

# 版本信息
VERSION     := $(shell git describe --tags --always --dirty)
COMMIT      := $(shell git rev-parse --short HEAD)
BUILD_TIME  := $(shell date -u '+%Y-%m-%dT%H:%M:%SZ')

# LDFLAGS
LDFLAGS := -s -w \
    -X main.version=$(VERSION) \
    -X main.commit=$(COMMIT) \
    -X main.buildTime=$(BUILD_TIME)

# 构建标志
GOFLAGS := -trimpath
CGO_ENABLED := 0

# 最优构建
.PHONY: build-optimized
build-optimized:
	CGO_ENABLED=$(CGO_ENABLED) GOOS=linux GOARCH=amd64 \
		go build $(GOFLAGS) -ldflags "$(LDFLAGS)" \
		-o bin/server ./cmd/server
	@echo "Binary size: $$(du -h bin/server | cut -f1)"

# UPX 压缩（可选，牺牲启动速度换取更小体积）
.PHONY: build-compressed
build-compressed: build-optimized
	upx --best bin/server
	@echo "Compressed size: $$(du -h bin/server | cut -f1)"

# 多架构构建
.PHONY: build-multiarch
build-multiarch:
	@for arch in amd64 arm64; do \
		echo "Building for linux/$$arch..."; \
		CGO_ENABLED=0 GOOS=linux GOARCH=$$arch \
			go build $(GOFLAGS) -ldflags "$(LDFLAGS)" \
			-o bin/server-linux-$$arch ./cmd/server; \
	done
```

## 🎯 面试题精选

### 1. 如何选择合适的构建工具？

**参考答案**：
选择构建工具需要考虑：
- **项目规模**：小型项目用 Make，大型单体仓库用 Bazel
- **语言生态**：Go 用 go build，Java 用 Gradle/Maven，Node.js 用 npm/pnpm
- **团队熟悉度**：选择团队熟悉的工具
- **构建速度**：pnpm > yarn > npm，Bazel > Make（大规模场景）
- **缓存需求**：Bazel/Gradle 支持远程缓存

推荐策略：用各语言原生工具构建，Make 作为统一入口，CI 平台处理编排。

### 2. npm、yarn、pnpm 有什么区别？

**参考答案**：
- **npm**：Node.js 默认，最广泛使用，但有幽灵依赖问题
- **yarn**：Meta 开发，支持 PnP（Plug'n'Play），速度快于 npm
- **pnpm**：使用硬链接和符号链接，解决幽灵依赖，磁盘占用最低，速度最快

pnpm 是当前推荐选择，特别是在单体仓库场景中。它通过硬链接共享依赖，避免了 node_modules 的重复存储。

### 3. 什么是幽灵依赖（Phantom Dependencies）？

**参考答案**：
幽灵依赖是指项目中使用了未在 package.json 中声明的依赖。这是因为 npm/yarn 的扁平化 node_modules 结构允许访问间接依赖。

例如：项目依赖 A，A 依赖 B。即使项目没有声明依赖 B，代码中也可以 `require('B')`。当 A 不再依赖 B 时，项目就会崩溃。

pnpm 通过严格的 node_modules 结构（只允许访问声明的依赖）解决了这个问题。

### 4. 如何在 CI/CD 中有效使用缓存？

**参考答案**：
- **Key 策略**：基于 lock 文件的 hash（而非分支），确保依赖变化时更新缓存
- **缓存路径**：只缓存必要目录（依赖目录，不包括构建产物）
- **缓存层级**：依赖缓存 > 构建缓存 > 测试缓存
- **远程缓存**：Bazel/Gradle 支持远程缓存，团队共享
- **Docker 缓存**：使用 BuildKit 缓存，按层缓存
- **失效策略**：定期清理过期缓存，避免缓存膨胀

### 5. Bazel 相比 Make 有什么优势？

**参考答案**：
- **精确依赖分析**：Bazel 构建完整的依赖图，只重建变更影响的部分
- **可重复性**：沙箱隔离确保构建结果一致
- **远程缓存**：支持团队级构建缓存共享
- **远程执行**：支持分布式构建
- **多语言支持**：统一的构建规则
- **大规模友好**：专为单体仓库设计

但 Bazel 学习曲线陡峭，小型项目不需要。Make 简单直接，适合大多数项目。

### 6. 如何优化 Docker 镜像构建速度？

**参考答案**：
- **多阶段构建**：分离构建环境和运行环境
- **层缓存**：先复制依赖文件，再复制源码（利用层缓存）
- **BuildKit**：启用 BuildKit 并行构建
- **最小基础镜像**：使用 alpine 或 distroless
- **.dockerignore**：排除不需要的文件
- **CI 缓存**：使用 `cache-from` 和 `cache-to`
