#### 1.2 完整的 Web 应用部署 Playbook

```yaml
# playbooks/deploy-webapp.yml
---
- name: Deploy Web Application
  hosts: webservers
  become: yes
  serial: "50%"    # 滚动更新，每次 50% 的主机

  vars:
    app_name: mywebapp
    app_version: "{{ lookup('env', 'APP_VERSION') | default('latest') }}"
    app_repo: "https://github.com/example/{{ app_name }}.git"
    app_dir: /opt/{{ app_name }}
    releases_dir: "{{ app_dir }}/releases"
    current_dir: "{{ app_dir }}/current"
    shared_dir: "{{ app_dir }}/shared"
    deploy_user: deploy
    deploy_group: deploy
    keep_releases: 5

  pre_tasks:
    - name: Verify deployment prerequisites
      assert:
        that:
          - app_version is defined and app_version | length > 0
          - ansible_memtotal_mb > 512
        fail_msg: "Deployment prerequisites not met"

    - name: Create deployment directory structure
      file:
        path: "{{ item }}"
        state: directory
        owner: "{{ deploy_user }}"
        group: "{{ deploy_group }}"
        mode: "0755"
      loop:
        - "{{ app_dir }}"
        - "{{ releases_dir }}"
        - "{{ shared_dir }}"
        - "{{ shared_dir }}/config"
        - "{{ shared_dir }}/logs"
        - "{{ shared_dir }}/uploads"

    - name: Start deployment
      debug:
        msg: "Deploying {{ app_name }} v{{ app_version }} to {{ inventory_hostname }}"

  tasks:
    - name: Clone new release
      git:
        repo: "{{ app_repo }}"
        dest: "{{ releases_dir }}/{{ app_version }}"
        version: "{{ app_version }}"
        force: yes
      become_user: "{{ deploy_user }}"
      register: clone_result

    - name: Install application dependencies
      pip:
        requirements: "{{ releases_dir }}/{{ app_version }}/requirements.txt"
        virtualenv: "{{ releases_dir }}/{{ app_version }}/venv"
        virtualenv_command: python3 -m venv
      become_user: "{{ deploy_user }}"
      when: clone_result.changed

    - name: Deploy application configuration
      template:
        src: templates/app.conf.j2
        dest: "{{ shared_dir }}/config/app.conf"
        owner: "{{ deploy_user }}"
        group: "{{ deploy_group }}"
        mode: "0640"
      notify: Restart application

    - name: Create shared symlinks
      file:
        src: "{{ shared_dir }}/{{ item }}"
        dest: "{{ releases_dir }}/{{ app_version }}/{{ item }}"
        state: link
        owner: "{{ deploy_user }}"
        force: yes
      loop:
        - config
        - logs
        - uploads

    - name: Run database migrations
      command: >
        {{ releases_dir }}/{{ app_version }}/venv/bin/python
        {{ releases_dir }}/{{ app_version }}/manage.py migrate --noinput
      args:
        chdir: "{{ releases_dir }}/{{ app_version }}"
      environment:
        APP_CONFIG: "{{ shared_dir }}/config/app.conf"
      become_user: "{{ deploy_user }}"
      register: migration_result
      when: clone_result.changed

    - name: Collect static files
      command: >
        {{ releases_dir }}/{{ app_version }}/venv/bin/python
        {{ releases_dir }}/{{ app_version }}/manage.py collectstatic --noinput
      args:
        chdir: "{{ releases_dir }}/{{ app_version }}"
      become_user: "{{ deploy_user }}"
      when: clone_result.changed

    - name: Update symlink to current release
      file:
        src: "{{ releases_dir }}/{{ app_version }}"
        dest: "{{ current_dir }}"
        state: link
        owner: "{{ deploy_user }}"
        force: yes

    - name: Restart application
      systemd:
        name: "{{ app_name }}"
        state: restarted
        daemon_reload: yes

    - name: Wait for application to be ready
      uri:
        url: "http://localhost:{{ app_port | default(8080) }}/health"
        method: GET
        status_code: 200
        timeout: 5
      register: health_check
      until: health_check.status == 200
      retries: 10
      delay: 5

    - name: Verify application version
      uri:
        url: "http://localhost:{{ app_port | default(8080) }}/version"
        method: GET
        return_content: yes
      register: version_check

    - name: Display deployment result
      debug:
        msg: |
          Deployment successful!
          Version deployed: {{ app_version }}
          Version running: {{ version_check.content | default('unknown') }}
          Host: {{ inventory_hostname }}

  post_tasks:
    - name: Clean old releases
      shell: |
        cd {{ releases_dir }}
        ls -t | tail -n +{{ keep_releases + 1 }} | xargs -r rm -rf
      become_user: "{{ deploy_user }}"
      changed_when: false

    - name: Record deployment
      lineinfile:
        path: /var/log/deployments.log
        line: "{{ ansible_date_time.iso8601 }} {{ inventory_hostname }} {{ app_name }} {{ app_version }} SUCCESS"
        create: yes
        mode: "0644"

  handlers:
    - name: Restart application
      systemd:
        name: "{{ app_name }}"
        state: restarted
```

#### 1.3 Rollback Playbook

```yaml
# playbooks/rollback-webapp.yml
---
- name: Rollback Web Application
  hosts: webservers
  become: yes
  serial: "100%"

  vars:
    app_name: mywebapp
    app_dir: /opt/{{ app_name }}
    releases_dir: "{{ app_dir }}/releases"
    current_dir: "{{ app_dir }}/current"
    deploy_user: deploy

  tasks:
    - name: Get current release version
      command: readlink -f {{ current_dir }}
      register: current_release
      changed_when: false

    - name: Get previous release version
      shell: |
        ls -t {{ releases_dir }} | sed -n '2p'
      register: previous_release
      changed_when: false

    - name: Verify previous release exists
      assert:
        that:
          - previous_release.stdout | length > 0
        fail_msg: "No previous release found for rollback"

    - name: Rollback to previous release
      file:
        src: "{{ releases_dir }}/{{ previous_release.stdout }}"
        dest: "{{ current_dir }}"
        state: link
        owner: "{{ deploy_user }}"
        force: yes

    - name: Restart application
      systemd:
        name: "{{ app_name }}"
        state: restarted
        daemon_reload: yes

    - name: Wait for application to be ready
      uri:
        url: "http://localhost:{{ app_port | default(8080) }}/health"
        method: GET
        status_code: 200
      register: health
      until: health.status == 200
      retries: 10
      delay: 5

    - name: Display rollback result
      debug:
        msg: |
          Rollback successful!
          From: {{ current_release.stdout | basename }}
          To: {{ previous_release.stdout }}
          Host: {{ inventory_hostname }}

    - name: Record rollback
      lineinfile:
        path: /var/log/deployments.log
        line: "{{ ansible_date_time.iso8601 }} {{ inventory_hostname }} {{ app_name }} ROLLBACK {{ previous_release.stdout }}"
        create: yes
```

### 2. 滚动更新

#### 2.1 serial 参数详解

```yaml
# 滚动更新策略
- name: Rolling update
  hosts: webservers
  serial: 1              # 每次更新 1 台主机
  # serial: "25%"        # 每次更新 25% 的主机
  # serial: [1, 25%, 100%]  # 先更新 1 台，再 25%，最后全部

  tasks:
    - name: Update application
      # ... 部署任务 ...

    - name: Health check
      uri:
        url: "http://localhost:{{ app_port }}/health"
        status_code: 200
      retries: 5
      delay: 10
```

#### 2.2 max_fail_percentage

```yaml
- name: Rolling update with failure tolerance
  hosts: webservers
  serial: "50%"
  max_fail_percentage: 20    # 允许 20% 的主机失败

  tasks:
    - name: Deploy and verify
      # ... 部署任务 ...
```

### 3. 蓝绿部署

#### 3.1 蓝绿部署 Playbook

```yaml
# playbooks/blue-green-deploy.yml
---
- name: Blue-Green Deployment
  hosts: localhost
  connection: local
  gather_facts: no

  vars:
    app_name: mywebapp
    app_version: "{{ lookup('env', 'APP_VERSION') | default('latest') }}"
    lb_config: /etc/nginx/conf.d/{{ app_name }}.conf
    active_env: "{{ lookup('file', '/opt/{{ app_name }}/active_env') | default('blue') }}"
    blue_servers:
      - web01:8080
      - web02:8080
    green_servers:
      - web03:8080
      - web04:8080

  tasks:
    - name: Determine target environment
      set_fact:
        target_env: "{{ 'green' if active_env == 'blue' else 'blue' }}"
        target_servers: "{{ green_servers if active_env == 'blue' else blue_servers }}"

    - name: Display deployment plan
      debug:
        msg: |
          Current active: {{ active_env }}
          Deploying to: {{ target_env }}
          Target servers: {{ target_servers }}

    - name: Deploy to target environment
      include_tasks: deploy_to_env.yml
      vars:
        target_hosts: "{{ target_env }}_servers"
        env_name: "{{ target_env }}"

    - name: Run health checks on target environment
      include_tasks: health_check.yml
      vars:
        servers: "{{ target_servers }}"

    - name: Switch load balancer to target environment
      template:
        src: templates/nginx_upstream.j2
        dest: "{{ lb_config }}"
        validate: "nginx -t -c /etc/nginx/nginx.conf"
      vars:
        upstream_servers: "{{ target_servers }}"

    - name: Reload Nginx
      service:
        name: nginx
        state: reloaded

    - name: Update active environment marker
      copy:
        content: "{{ target_env }}"
        dest: "/opt/{{ app_name }}/active_env"
        mode: "0644"

    - name: Display deployment result
      debug:
        msg: |
          Deployment successful!
          Active environment: {{ target_env }}
          Version: {{ app_version }}
```

```yaml
# roles/postgresql/tasks/install.yml
---
- name: Install PostgreSQL packages
  package:
    name:
      - postgresql
      - postgresql-server
      - postgresql-contrib
      - python3-psycopg2
    state: present

- name: Initialize PostgreSQL (RedHat)
  command: postgresql-setup --initdb
  args:
    creates: /var/lib/pgsql/data/PG_VERSION
  when: ansible_os_family == "RedHat"

- name: Start and enable PostgreSQL
  service:
    name: postgresql
    state: started
    enabled: yes
```

```yaml
# roles/postgresql/tasks/configure.yml
---
- name: Configure postgresql.conf
  template:
    src: postgresql.conf.j2
    dest: "{{ pg_config_dir }}/postgresql.conf"
    owner: postgres
    group: postgres
    mode: "0644"
  notify: Restart postgresql

- name: Configure pg_hba.conf
  template:
    src: pg_hba.conf.j2
    dest: "{{ pg_config_dir }}/pg_hba.conf"
    owner: postgres
    group: postgres
    mode: "0640"
  notify: Restart postgresql
```

```yaml
# roles/postgresql/tasks/databases.yml
---
- name: Create application database
  postgresql_db:
    name: "{{ app_db_name }}"
    encoding: UTF-8
    lc_collate: en_US.UTF-8
    lc_ctype: en_US.UTF-8
    template: template0
    state: present
  become: yes
  become_user: postgres

- name: Create application database user
  postgresql_user:
    name: "{{ app_db_user }}"
    password: "{{ app_db_password }}"
    db: "{{ app_db_name }}"
    priv: "ALL"
    state: present
  become: yes
  become_user: postgres
  no_log: true

- name: Create read-only user for reporting
  postgresql_user:
    name: "{{ app_db_readonly_user }}"
    password: "{{ app_db_readonly_password }}"
    db: "{{ app_db_name }}"
    priv: "SELECT"
    state: present
  become: yes
  become_user: postgres
  no_log: true

- name: Run database migrations
  command: >
    {{ app_dir }}/venv/bin/python
    {{ app_dir }}/current/manage.py migrate --noinput
  args:
    chdir: "{{ app_dir }}/current"
  environment:
    DATABASE_URL: "postgres://{{ app_db_user }}:{{ app_db_password }}@localhost/{{ app_db_name }}"
  become_user: "{{ deploy_user }}"
  register: migration_result
  changed_when: "'No migrations to apply' not in migration_result.stdout"
```

#### 4.2 数据库备份 Role

```yaml
# roles/postgresql/tasks/backup.yml
---
- name: Create backup directory
  file:
    path: "{{ pg_backup_dir }}"
    state: directory
    owner: postgres
    group: postgres
    mode: "0750"

- name: Deploy backup script
  template:
    src: pg_backup.sh.j2
    dest: /usr/local/bin/pg_backup.sh
    owner: postgres
    group: postgres
    mode: "0750"

- name: Deploy restore script
  template:
    src: pg_restore.sh.j2
    dest: /usr/local/bin/pg_restore.sh
    owner: postgres
    group: postgres
    mode: "0750"

- name: Schedule daily backup
  cron:
    name: "PostgreSQL daily backup"
    minute: "0"
    hour: "2"
    user: postgres
    job: "/usr/local/bin/pg_backup.sh >> {{ pg_backup_dir }}/backup.log 2>&1"
    state: present

- name: Schedule weekly backup rotation
  cron:
    name: "PostgreSQL backup rotation"
    minute: "0"
    hour: "3"
    weekday: "0"
    user: postgres
    job: "find {{ pg_backup_dir }} -name '*.sql.gz' -mtime +{{ pg_backup_retention_days | default(30) }} -delete"
    state: present
```

```bash
# roles/postgresql/templates/pg_backup.sh.j2
#!/bin/bash
# PostgreSQL Backup Script - Managed by Ansible

set -euo pipefail

BACKUP_DIR="{{ pg_backup_dir }}"
TIMESTAMP=$(date +%Y%m%d_%H%M%S)
DATABASES="{{ app_db_name }}"
RETENTION_DAYS="{{ pg_backup_retention_days | default(30) }}"

echo "=== PostgreSQL Backup Started: $(date) ==="

for DB in $DATABASES; do
    BACKUP_FILE="${BACKUP_DIR}/${DB}_${TIMESTAMP}.sql.gz"
    echo "Backing up database: ${DB}"
    pg_dump -U postgres -d "${DB}" | gzip > "${BACKUP_FILE}"
    echo "Backup saved: ${BACKUP_FILE} ($(du -h "${BACKUP_FILE}" | cut -f1))"
done

# Clean old backups
find "${BACKUP_DIR}" -name "*.sql.gz" -mtime +${RETENTION_DAYS} -delete
echo "Cleaned backups older than ${RETENTION_DAYS} days"

echo "=== PostgreSQL Backup Completed: $(date) ==="
```

## 🎯 面试题精选

### 1. 滚动更新和蓝绿部署有什么区别？各自的优缺点是什么？

**参考答案：**
- **滚动更新**：逐步更新服务器，始终保持部分服务器可用。优点是资源占用少，缺点是回滚较慢。
- **蓝绿部署**：维护两套完整环境，通过切换负载均衡器实现更新。优点是回滚极快（切换 LB），缺点是需要双倍资源。

Ansible 实现滚动更新使用 `serial` 参数，蓝绿部署使用 `pre_tasks` + `roles` + `post_tasks` 编排。

### 2. 如何在 Ansible 中实现零停机部署？

**参考答案：**
1. 使用 `serial` 参数实现滚动更新
2. 在每批更新后执行健康检查（`uri` 模块）
3. 使用 `max_fail_percentage` 容忍少量失败
4. 使用 release 目录 + symlink 模式部署
5. 数据库迁移在切换 symlink 之前执行
6. 使用 handler 在所有任务完成后重启服务

### 3. 如何处理部署过程中的数据库迁移？

**参考答案：**
1. 在切换应用版本之前执行迁移
2. 确保迁移是向前兼容的（新旧版本都能运行）
3. 使用 `register` 记录迁移结果
4. 如果迁移失败，使用 `block/rescue` 回滚
5. 大型迁移使用异步执行（`async` + `poll`）

### 4. 如何将 Ansible 集成到 CI/CD 流水线？

**参考答案：**
1. 在 CI 中运行 `ansible-lint` 和 `--syntax-check`
2. 使用 CI 环境变量传递敏感信息（Vault 密码、SSH 密钥）
3. 使用 `--vault-password-file` 和环境变量传递 Vault 密码
4. 设置 `ANSIBLE_HOST_KEY_CHECKING=False` 跳过主机密钥检查
5. 使用 `when: manual` 实现手动触发生产部署

### 5. 如何实现部署回滚？

**参考答案：**
1. 使用 release 目录模式保留多个版本
2. 回滚时仅切换 symlink 到上一个版本
3. 保留数据库迁移脚本，必要时执行回滚迁移
4. 使用 `readlink` 获取当前版本，`ls -t | sed -n '2p'` 获取上一个版本
5. 回滚后重启服务并执行健康检查
