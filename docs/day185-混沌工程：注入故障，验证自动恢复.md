### 2. Chaos Mesh 安装

```bash
#!/usr/bin/env bash
# scripts/install-chaos-mesh.sh

set -euo pipefail

NAMESPACE="chaos-mesh"

echo "=== 安装 Chaos Mesh ==="
helm repo add chaos-mesh https://charts.chaos-mesh.org
helm repo update

kubectl create namespace "${NAMESPACE}" --dry-run=client -o yaml | kubectl apply -f -

helm install chaos-mesh chaos-mesh/chaos-mesh \
  --namespace "${NAMESPACE}" \
  --set dashboard.securityMode=true \
  --set controllerManager.replicaCount=1 \
  --set chaosDaemon.runtime=containerd \
  --set chaosDaemon.socketPath=/run/containerd/containerd.sock \
  --wait \
  --timeout 5m

echo "=== 等待 Chaos Mesh 就绪 ==="
kubectl wait --for=condition=ready pod -l app.kubernetes.io/name=chaos-mesh \
  -n "${NAMESPACE}" --timeout=120s

echo "=== 获取 Dashboard 地址 ==="
echo "kubectl port-forward svc/chaos-dashboard 2333:2333 -n ${NAMESPACE}"
echo "访问: http://localhost:2333"

echo "=== 安装 chaosctl ==="
curl -sSL https://mirrors.chaos-mesh.org/latest/chaosctl-linux-amd64.tar.gz | tar xz
sudo mv linux-amd64/chaosctl /usr/local/bin/
rm -rf linux-amd64
```

### 4. 自动恢复验证脚本

```bash
#!/usr/bin/env bash
# chaos/scripts/verify-recovery.sh
# 验证系统自动恢复能力

set -euo pipefail

NAMESPACE="sre-capstone"
MAX_WAIT=300  # 最大等待 5 分钟
CHECK_INTERVAL=10  # 每 10 秒检查一次

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

log_info() {
    echo -e "${GREEN}[INFO]${NC} $1"
}

log_warn() {
    echo -e "${YELLOW}[WARN]${NC} $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

# ============================================================
# 检查 Pod 状态
# ============================================================
check_pods_ready() {
    local deployment=$1
    local expected_replicas=$2

    local ready_replicas
    ready_replicas=$(kubectl get deployment "${deployment}" -n "${NAMESPACE}" \
        -o jsonpath='{.status.readyReplicas}' 2>/dev/null || echo "0")

    if [[ "${ready_replicas}" == "${expected_replicas}" ]]; then
        return 0
    else
        return 1
    fi
}

# ============================================================
# 等待 Pod 恢复
# ============================================================
wait_for_recovery() {
    local deployment=$1
    local expected_replicas=$2
    local start_time
    start_time=$(date +%s)

    log_info "等待 ${deployment} 恢复到 ${expected_replicas} 个副本..."

    while true; do
        local current_time
        current_time=$(date +%s)
        local elapsed=$((current_time - start_time))

        if [[ ${elapsed} -ge ${MAX_WAIT} ]]; then
            log_error "${deployment} 在 ${MAX_WAIT} 秒内未恢复"
            return 1
        fi

        if check_pods_ready "${deployment}" "${expected_replicas}"; then
            log_info "${deployment} 已恢复 (${elapsed}秒)"
            return 0
        fi

        local ready
        ready=$(kubectl get deployment "${deployment}" -n "${NAMESPACE}" \
            -o jsonpath='{.status.readyReplicas}' 2>/dev/null || echo "0")
        log_warn "${deployment} 等待中... (${ready}/${expected_replicas}) ${elapsed}秒"

        sleep ${CHECK_INTERVAL}
    done
}

# ============================================================
# 验证 API 可用性
# ============================================================
verify_api_availability() {
    local service=$1
    local max_retries=5
    local retry=0

    log_info "验证 ${service} API 可用性..."

    while [[ ${retry} -lt ${max_retries} ]]; do
        local response
        response=$(kubectl exec -n "${NAMESPACE}" deploy/user-service -- \
            wget -qO- "http://${service}/health" 2>/dev/null || echo "")

        if echo "${response}" | grep -q "healthy"; then
            log_info "${service} API 可用"
            return 0
        fi

        retry=$((retry + 1))
        sleep 5
    done

    log_error "${service} API 不可用"
    return 1
}

# ============================================================
# 主验证流程
# ============================================================
main() {
    log_info "=========================================="
    log_info "  混沌工程 - 自动恢复验证"
    log_info "=========================================="

    local all_passed=true

    # 1. 记录初始状态
    log_info "Step 1: 记录初始状态"
    kubectl get pods -n "${NAMESPACE}" -o wide
    echo ""

    # 2. 等待 Pod Kill 恢复
    log_info "Step 2: 验证 Pod Kill 恢复"
    if wait_for_recovery "user-service" "2"; then
        log_info "Pod Kill 恢复验证通过"
    else
        log_error "Pod Kill 恢复验证失败"
        all_passed=false
    fi
    echo ""

    # 3. 等待 Network Delay 恢复
    log_info "Step 3: 验证网络延迟恢复"
    if wait_for_recovery "product-service" "2"; then
        log_info "网络延迟恢复验证通过"
    else
        log_error "网络延迟恢复验证失败"
        all_passed=false
    fi
    echo ""

    # 4. 验证 API 可用性
    log_info "Step 4: 验证 API 可用性"
    if verify_api_availability "user-service"; then
        log_info "User Service API 验证通过"
    else
        log_error "User Service API 验证失败"
        all_passed=false
    fi

    if verify_api_availability "product-service"; then
        log_info "Product Service API 验证通过"
    else
        log_error "Product Service API 验证失败"
        all_passed=false
    fi
    echo ""

    # 5. 检查 HPA 状态
    log_info "Step 5: 检查 HPA 状态"
    kubectl get hpa -n "${NAMESPACE}"
    echo ""

    # 6. 总结
    log_info "=========================================="
    if [[ "${all_passed}" == "true" ]]; then
        log_info "所有恢复验证通过!"
    else
        log_error "部分恢复验证失败，请检查系统状态"
    fi
    log_info "=========================================="
}

main "$@"
```

### 5. 告警验证脚本

```bash
#!/usr/bin/env bash
# chaos/scripts/verify-alerts.sh
# 验证告警系统是否正确触发

set -euo pipefail

NAMESPACE="observability"
PROMETHEUS_URL="http://localhost:9090"

log_info() {
    echo -e "\033[0;32m[INFO]\033[0m $1"
}

log_warn() {
    echo -e "\033[1;33m[WARN]\033[0m $1"
}

# ============================================================
# 检查告警状态
# ============================================================
check_alert_firing() {
    local alert_name=$1

    local status
    status=$(curl -s "${PROMETHEUS_URL}/api/v1/alerts" | \
        jq -r ".data.alerts[] | select(.labels.alertname==\"${alert_name}\") | .state" 2>/dev/null)

    if [[ "${status}" == "firing" ]]; then
        log_info "告警 ${alert_name} 已触发"
        return 0
    else
        log_warn "告警 ${alert_name} 未触发 (状态: ${status:-unknown})"
        return 1
    fi
}

# ============================================================
# 主验证流程
# ============================================================
main() {
    log_info "=========================================="
    log_info "  告警系统验证"
    log_info "=========================================="

    # 启动 port-forward
    kubectl port-forward svc/prometheus-kube-prometheus-prometheus 9090:9090 \
        -n "${NAMESPACE}" &
    PF_PID=$!
    sleep 5

    local alerts_to_check=(
        "PodCrashLooping"
        "PodNotReady"
        "HighErrorRate"
        "ServiceDown"
    )

    for alert in "${alerts_to_check[@]}"; do
        check_alert_firing "${alert}" || true
    done

    # 清理
    kill ${PF_PID} 2>/dev/null || true
}

main "$@"
```

### 6. 事故响应演练

```bash
#!/usr/bin/env bash
# chaos/scripts/incident-response-drill.sh
# 事故响应演练脚本

set -euo pipefail

NAMESPACE="sre-capstone"
SLACK_WEBHOOK="${SLACK_WEBHOOK_URL:-}"

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

log_info() {
    echo -e "${GREEN}[INFO]${NC} $1"
}

log_warn() {
    echo -e "${YELLOW}[DRILL]${NC} $1"
}

log_error() {
    echo -e "${RED}[INCIDENT]${NC} $1"
}

send_slack_notification() {
    local message=$1
    if [[ -n "${SLACK_WEBHOOK}" ]]; then
        curl -s -X POST "${SLACK_WEBHOOK}" \
            -H "Content-Type: application/json" \
            -d "{\"text\": \"[DRILL] ${message}\"}" > /dev/null
    fi
}

# ============================================================
# 演练阶段 1: 故障注入
# ============================================================
phase_1_inject() {
    log_warn "=========================================="
    log_warn "  阶段 1: 故障注入"
    log_warn "=========================================="

    send_slack_notification "事故响应演练开始: 注入 Pod Kill 故障"

    # 注入 Pod Kill
    kubectl apply -f chaos/experiments/pod-kill.yaml
    log_warn "已注入 Pod Kill 故障"

    # 等待故障生效
    sleep 30

    # 检查 Pod 状态
    log_warn "当前 Pod 状态:"
    kubectl get pods -n "${NAMESPACE}" -o wide
}

# ============================================================
# 演练阶段 2: 检测与告警
# ============================================================
phase_2_detect() {
    log_warn "=========================================="
    log_warn "  阶段 2: 检测与告警"
    log_warn "=========================================="

    send_slack_notification "事故响应演练: 检测阶段"

    # 检查告警
    log_warn "检查 Prometheus 告警..."
    kubectl port-forward svc/prometheus-kube-prometheus-prometheus 9090:9090 \
        -n observability &
    PF_PID=$!
    sleep 5

    local alerts
    alerts=$(curl -s http://localhost:9090/api/v1/alerts | jq '.data.alerts | length')
    log_warn "当前活跃告警数: ${alerts}"

    kill ${PF_PID} 2>/dev/null || true
}

# ============================================================
# 演练阶段 3: 响应与恢复
# ============================================================
phase_3_respond() {
    log_warn "=========================================="
    log_warn "  阶段 3: 响应与恢复"
    log_warn "=========================================="

    send_slack_notification "事故响应演练: 响应阶段"

    # 观察自动恢复
    log_warn "观察自动恢复过程..."

    local max_wait=180
    local elapsed=0

    while [[ ${elapsed} -lt ${max_wait} ]]; do
        local ready
        ready=$(kubectl get deployment user-service -n "${NAMESPACE}" \
            -o jsonpath='{.status.readyReplicas}' 2>/dev/null || echo "0")

        if [[ "${ready}" == "2" ]]; then
            log_info "User Service 已恢复到 2 个副本"
            break
        fi

        log_warn "等待恢复... (${ready}/2) ${elapsed}秒"
        sleep 10
        elapsed=$((elapsed + 10))
    done

    if [[ ${elapsed} -ge ${max_wait} ]]; then
        log_error "自动恢复超时，需要人工干预"
        send_slack_notification "事故响应演练: 自动恢复超时，需要人工干预"
    fi
}

# ============================================================
# 演练阶段 4: 验证恢复
# ============================================================
phase_4_verify() {
    log_warn "=========================================="
    log_warn "  阶段 4: 验证恢复"
    log_warn "=========================================="

    send_slack_notification "事故响应演练: 验证阶段"

    # 验证所有 Pod 正常
    log_warn "验证 Pod 状态..."
    kubectl get pods -n "${NAMESPACE}"

    # 验证 API 可用性
    log_warn "验证 API 可用性..."
    local response
    response=$(kubectl exec -n "${NAMESPACE}" deploy/user-service -- \
        wget -qO- http://user-service/health 2>/dev/null || echo "failed")

    if echo "${response}" | grep -q "healthy"; then
        log_info "API 验证通过"
    else
        log_error "API 验证失败"
    fi

    # 验证 HPA 状态
    log_warn "验证 HPA 状态..."
    kubectl get hpa -n "${NAMESPACE}"
}

# ============================================================
# 演练阶段 5: 清理与总结
# ============================================================
phase_5_cleanup() {
    log_warn "=========================================="
    log_warn "  阶段 5: 清理与总结"
    log_warn "=========================================="

    # 清理混沌实验
    kubectl delete -f chaos/experiments/pod-kill.yaml --ignore-not-found

    send_slack_notification "事故响应演练完成"

    log_info "演练完成! 请填写以下复盘信息:"
    echo ""
    echo "1. 故障检测时间: ____秒"
    echo "2. 自动恢复时间: ____秒"
    echo "3. 是否需要人工干预: 是/否"
    echo "4. 发现的问题: ____"
    echo "5. 改进建议: ____"
}

# ============================================================
# 主流程
# ============================================================
main() {
    log_warn "=========================================="
    log_warn "  事故响应演练开始"
    log_warn "=========================================="
    echo ""
    log_warn "本次演练将模拟以下场景:"
    log_warn "1. Pod Kill 故障注入"
    log_warn "2. 告警触发验证"
    log_warn "3. 自动恢复观察"

    phase_1_inject
    phase_2_detect
    phase_3_respond
    phase_4_verify
    phase_5_cleanup
}

main "$@"
```

## 🎯 面试题精选

### 问题 1：什么是混沌工程？它的价值是什么？

**参考答案**：

混沌工程是通过主动注入故障来发现系统弱点的实践。

价值：
1. **发现隐藏问题**：在生产环境之前发现潜在故障
2. **验证恢复能力**：确认自动恢复机制是否工作
3. **建立信心**：团队对系统稳定性更有信心
4. **改进监控**：发现监控盲点
5. **优化架构**：推动架构改进

### 问题 2：如何设计合理的混沌实验？

**参考答案**：

1. **从稳态开始**：定义系统的正常状态指标
2. **小范围实验**：先在测试环境，再逐步扩大
3. **可控故障**：故障注入要有明确的范围和时间
4. **观察指标**：监控所有相关指标
5. **可回滚**：实验失败时能快速回滚
6. **记录结果**：记录每次实验的结果和发现

### 问题 3：常见的故障注入类型有哪些？

**参考答案**：

| 类型 | 描述 | 影响 |
|------|------|------|
| Pod Kill | 杀死容器 | 测试自动重启 |
| Network Delay | 增加网络延迟 | 测试超时处理 |
| Network Partition | 网络分区 | 测试容错能力 |
| CPU Stress | CPU 压力 | 测试资源限制 |
| Memory Stress | 内存压力 | 测试 OOM 处理 |
| DNS Failure | DNS 故障 | 测试服务发现 |
| IO Delay | IO 延迟 | 测试存储容错 |

### 问题 4：事故响应的标准流程是什么？

**参考答案**：

1. **检测**：监控系统发现异常
2. **告警**：AlertManager 发送通知
3. **响应**：On-Call 工程师响应
4. **诊断**：使用可观测性工具定位根因
5. **修复**：实施修复措施
6. **恢复**：验证系统恢复
7. **复盘**：编写 RCA 报告，记录改进项

### 问题 5：如何衡量混沌工程的效果？

**参考答案**：

1. **故障检测时间 (MTTD)**：从故障发生到检测的时间
2. **故障恢复时间 (MTTR)**：从检测到恢复的时间
3. **告警准确率**：有效告警占总告警的比例
4. **自动恢复率**：无需人工干预的恢复比例
5. **发现的问题数**：每次实验发现的问题

### 问题 6：Chaos Mesh 和 Litmus 的区别？

**参考答案**：

| 特性 | Chaos Mesh | Litmus |
|------|-----------|--------|
| 平台 | K8s 原生 | K8s + 通用 |
| 实验类型 | 丰富 | 非常丰富 |
| UI | 内置 Dashboard | Litmus Portal |
| 社区 | CNCF 项目 | CNCF 项目 |
| 学习曲线 | 中等 | 较高 |

### 问题 7：混沌实验的安全边界在哪里？

**参考答案**：

安全边界：
1. **非生产环境优先**：先在测试环境验证
2. **灰度发布**：从小范围开始
3. **时间限制**：设置实验持续时间
4. **影响范围**：限制影响的 Pod/节点数量
5. **快速回滚**：随时可以停止实验
6. **监控告警**：实验期间加强监控
