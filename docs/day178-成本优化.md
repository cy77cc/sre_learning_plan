#### 2.2 资源 Right-sizing

```python
# resource_rightsizing.py

class ResourceRightsizer:
    def __init__(self):
        self.recommendations = []
    
    def analyze_instance(self, instance_id, metrics):
        """
        分析实例资源使用，生成 right-sizing 建议
        
        metrics: {
            'cpu_avg': 平均 CPU 使用率,
            'cpu_p95': P95 CPU 使用率,
            'memory_avg': 平均内存使用率,
            'memory_p95': P95 内存使用率,
            'current_type': 当前实例类型,
        }
        """
        cpu_avg = metrics['cpu_avg']
        cpu_p95 = metrics['cpu_p95']
        memory_avg = metrics['memory_avg']
        memory_p95 = metrics['memory_p95']
        
        # 判断是否过度配置
        if cpu_p95 < 40 and memory_p95 < 50:
            action = "downsize"
            reason = "CPU 和内存使用率都过低"
            savings = "30-50%"
        elif cpu_p95 < 50:
            action = "downsize_cpu"
            reason = "CPU 使用率过低"
            savings = "20-30%"
        elif memory_p95 < 50:
            action = "downsize_memory"
            reason = "内存使用率过低"
            savings = "20-30%"
        elif cpu_p95 > 80 or memory_p95 > 85:
            action = "upsize"
            reason = "资源使用率过高，有性能风险"
            savings = "N/A (增加成本)"
        else:
            action = "keep"
            reason = "资源配置合理"
            savings = "0%"
        
        recommendation = {
            'instance_id': instance_id,
            'current_type': metrics['current_type'],
            'cpu_avg': cpu_avg,
            'cpu_p95': cpu_p95,
            'memory_avg': memory_avg,
            'memory_p95': memory_p95,
            'action': action,
            'reason': reason,
            'estimated_savings': savings
        }
        
        self.recommendations.append(recommendation)
        return recommendation
    
    def generate_report(self):
        """生成 right-sizing 报告"""
        report = "# 资源 Right-sizing 报告\n\n"
        
        downsize = [r for r in self.recommendations if 'downsize' in r['action']]
        upsize = [r for r in self.recommendations if r['action'] == 'upsize']
        keep = [r for r in self.recommendations if r['action'] == 'keep']
        
        report += f"## 概览\n"
        report += f"- 建议缩容: {len(downsize)} 个实例\n"
        report += f"- 建议扩容: {len(upsize)} 个实例\n"
        report += f"- 保持不变: {len(keep)} 个实例\n\n"
        
        if downsize:
            report += "## 建议缩容的实例\n"
            for r in downsize:
                report += f"\n### {r['instance_id']}\n"
                report += f"- 当前类型: {r['current_type']}\n"
                report += f"- CPU P95: {r['cpu_p95']}%\n"
                report += f"- 内存 P95: {r['memory_p95']}%\n"
                report += f"- 建议: {r['action']}\n"
                report += f"- 原因: {r['reason']}\n"
                report += f"- 预计节省: {r['estimated_savings']}\n"
        
        return report

# 使用示例
sizer = ResourceRightsizer()

# 分析多个实例
instances = [
    ('i-001', {'cpu_avg': 15, 'cpu_p95': 25, 'memory_avg': 30, 'memory_p95': 40, 'current_type': 'c5.2xlarge'}),
    ('i-002', {'cpu_avg': 60, 'cpu_p95': 75, 'memory_avg': 70, 'memory_p95': 80, 'current_type': 'c5.xlarge'}),
    ('i-003', {'cpu_avg': 80, 'cpu_p95': 92, 'memory_avg': 75, 'memory_p95': 88, 'current_type': 'c5.xlarge'}),
]

for instance_id, metrics in instances:
    sizer.analyze_instance(instance_id, metrics)

print(sizer.generate_report())
```

### 6. FinOps 实践

#### 6.1 FinOps 框架

```
┌─────────────────────────────────────────────────────────┐
│                  FinOps 框架                              │
│                                                         │
│  FinOps 是一种云财务管理方法论，                          │
│  通过协作、数据驱动和持续优化来管理云成本。               │
│                                                         │
│  三阶段循环：                                            │
│  ┌──────────────────────────────────────────────┐       │
│  │                                              │       │
│  │   Inform → Optimize → Operate                │       │
│  │     ↑                        │               │       │
│  │     └────────────────────────┘               │       │
│  │                                              │       │
│  └──────────────────────────────────────────────┘       │
│                                                         │
│  Inform（通知）：                                        │
│  ┌───────────────────────────────────────┐              │
│  │ • 可见性：谁在用什么，花多少钱         │              │
│  │ • 分配：将成本分配到团队/项目          │              │
│  │ • 基准：建立成本基准和趋势             │              │
│  │ • 预测：预测未来成本                   │              │
│  └───────────────────────────────────────┘              │
│                                                         │
│  Optimize（优化）：                                      │
```

### 7. 成本告警

#### 7.1 成本告警配置

```python
# cost_alerts.py

class CostAlertManager:
    def __init__(self):
        self.alerts = []
        self.budgets = {}
    
    def set_budget(self, name, amount, period='monthly'):
        """设置预算"""
        self.budgets[name] = {
            'amount': amount,
            'period': period,
            'alerts': []
        }
    
    def add_alert(self, budget_name, threshold_pct, action='notify'):
        """添加告警规则"""
        if budget_name in self.budgets:
            self.budgets[budget_name]['alerts'].append({
                'threshold': threshold_pct,
                'action': action
            })
    
    def check_costs(self, current_costs):
        """检查成本是否触发告警"""
        triggered = []
        
        for budget_name, budget in self.budgets.items():
            if budget_name in current_costs:
                current = current_costs[budget_name]
                budget_amount = budget['amount']
                usage_pct = (current / budget_amount) * 100
                
                for alert in budget['alerts']:
                    if usage_pct >= alert['threshold']:
                        triggered.append({
                            'budget': budget_name,
                            'current': current,
                            'budget_amount': budget_amount,
                            'usage_pct': usage_pct,
                            'threshold': alert['threshold'],
                            'action': alert['action']
                        })
        
        return triggered

# 使用示例
manager = CostAlertManager()

# 设置预算
manager.set_budget('compute', 50000)  # 计算预算 5 万/月
manager.set_budget('storage', 10000)  # 存储预算 1 万/月
manager.set_budget('network', 5000)   # 网络预算 5 千/月

# 添加告警规则
manager.add_alert('compute', 80, 'notify')   # 80% 时通知
manager.add_alert('compute', 100, 'alert')   # 100% 时告警
manager.add_alert('compute', 120, 'block')   # 120% 时阻止

manager.add_alert('storage', 90, 'notify')
manager.add_alert('storage', 100, 'alert')

# 检查当前成本
current_costs = {
    'compute': 45000,
    'storage': 9500,
    'network': 3000
}

triggered = manager.check_costs(current_costs)

print("=== 成本告警检查 ===\n")
for alert in triggered:
    print(f"预算: {alert['budget']}")
    print(f"  当前: ${alert['current']:,.0f}")
    print(f"  预算: ${alert['budget_amount']:,.0f}")
    print(f"  使用率: {alert['usage_pct']:.1f}%")
    print(f"  触发阈值: {alert['threshold']}%")
    print(f"  动作: {alert['action']}")
    print()
```

```python
# cost_analysis.py

def analyze_monthly_costs(cost_data):
    """
    分析月度成本数据
    
    cost_data: {
        'EC2': 30000,
        'RDS': 15000,
        'S3': 5000,
        'CloudFront': 3000,
        'DataTransfer': 2000,
        'Other': 5000
    }
    """
    total = sum(cost_data.values())
    
    print("=== 月度成本分析报告 ===\n")
    print(f"总成本: ${total:,.0f}\n")
    
    print("各服务成本占比:")
    for service, cost in sorted(cost_data.items(), key=lambda x: x[1], reverse=True):
        pct = (cost / total) * 100
        print(f"  {service}: ${cost:,.0f} ({pct:.1f}%)")
    
    # 优化建议
    print("\n=== 优化建议 ===\n")
    
    suggestions = []
    
    # EC2 优化
    if cost_data.get('EC2', 0) > total * 0.4:
        suggestions.append({
            'service': 'EC2',
            'suggestion': '考虑购买 Savings Plans 或 RI',
            'potential_savings': '20-40%',
            'estimated_monthly': cost_data['EC2'] * 0.3
        })
    
    # S3 优化
    if cost_data.get('S3', 0) > 1000:
        suggestions.append({
            'service': 'S3',
            'suggestion': '实施存储分层策略',
            'potential_savings': '30-50%',
            'estimated_monthly': cost_data['S3'] * 0.3
        })
    
    # 数据传输优化
    if cost_data.get('DataTransfer', 0) > 1000:
        suggestions.append({
            'service': 'DataTransfer',
            'suggestion': '使用 CloudFront 或 VPC Endpoints',
            'potential_savings': '20-30%',
            'estimated_monthly': cost_data['DataTransfer'] * 0.2
        })
    
    total_savings = 0
    for s in suggestions:
        print(f"服务: {s['service']}")
        print(f"  建议: {s['suggestion']}")
        print(f"  预计节省: {s['potential_savings']}")
        print(f"  预计月省: ${s['estimated_monthly']:,.0f}")
        total_savings += s['estimated_monthly']
        print()
    
    print(f"=== 总计预计月省: ${total_savings:,.0f} ===")

# 使用示例
cost_data = {
    'EC2': 30000,
    'RDS': 15000,
    'S3': 5000,
    'CloudFront': 3000,
    'DataTransfer': 2000,
    'Other': 5000
}

analyze_monthly_costs(cost_data)
```

## 🎯 面试题精选

### 面试题 1：如何优化云成本？

**考察点**：对成本优化策略的全面理解。

**参考答案要点**：
1. Right-sizing：调整资源大小匹配实际需求
2. 预留容量：购买 RI 或 Savings Plans
3. Spot 实例：对无状态工作负载使用 Spot
4. 存储分层：根据访问频率使用不同存储类
5. 自动化：自动关闭非工作时间资源

### 面试题 2：RI 和 Savings Plans 有什么区别？

**考察点**：对预留容量选项的理解。

**参考答案要点**：
1. RI 指定具体实例类型，SP 承诺消费金额
2. SP 更灵活，可跨服务使用
3. RI 折扣可能更高，但灵活性低
4. 推荐根据工作负载特点选择

### 面试题 3：Spot 实例适合什么场景？

**考察点**：对 Spot 实例适用性的理解。

**参考答案要点**：
1. 适合：无状态、可中断、可重试的工作负载
2. 适合：批处理、CI/CD、数据分析
3. 不适合：有状态服务、数据库、核心交易
4. 需要处理中断（2分钟通知）

### 面试题 4：什么是 FinOps？

**考察点**：对 FinOps 方法论的理解。

**参考答案要点**：
1. 云财务管理方法论
2. 三阶段：Inform → Optimize → Operate
3. 核心理念：协作、数据驱动、持续优化
4. 目标：最大化云投资的商业价值

### 面试题 5：如何建立成本告警机制？

**考察点**：对成本管理实践的理解。

**参考答案要点**：
1. 设置预算和告警阈值
2. 按服务、团队、项目分配成本
3. 使用标签策略追踪成本
4. 定期审查和优化
